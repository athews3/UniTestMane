(* ===================================================== *)
(* ---------------- EDGE DETECTION --------------------- *)
(* ===================================================== *)
(*
    Purpose:
    The HMI buttons are physical/touchscreen signals that may stay ON
    for more than one PLC scan. The PLC scans this program repeatedly,
    many times per second.

    If we acted directly on the button being ON, then a technician pressing
    a button once could accidentally trigger the command many times.

    These "Rise" bits detect only the first scan where the command changes
    from OFF to ON.

    Example:
        HMI_StartStop = OFF last scan, ON this scan
        → StartStop_Rise = TRUE for one scan only

    This gives us one clean command per button press.
*)

StartStop_Rise := HMI_StartStop AND NOT PrevStartStop;
Hold_Rise      := HMI_Hold      AND NOT PrevHold;
OffHold_Rise   := HMI_OffHold   AND NOT PrevOffHold;

(*
    Store the current HMI button states so that on the next scan
    the PLC can tell whether the button just turned ON or was already ON.
*)

PrevStartStop := HMI_StartStop;
PrevHold      := HMI_Hold;
PrevOffHold   := HMI_OffHold;


(* ===================================================== *)
(* ---------------- HOLD REASON VALIDATION ------------- *)
(* ===================================================== *)
(*
    Purpose:
    If a technician places a station on hold, they must provide a reason.

    HoldReason_Code is expected to be a numeric value selected from the HMI.
    A value greater than zero means the technician selected a valid reason.
    A zero value means no reason has been selected.

    This makes the HMI/database record cleaner because every hold event
    can be tied to a reason code.
*)

HoldReason_Valid := (HoldReason_Code > 0);


(* ===================================================== *)
(* ---------------- SAFETY / INTERLOCK ----------------- *)
(* ===================================================== *)
(*
    Purpose:
    X17 is treated as the main safety/permissive input.

    Expected meaning:
        X17 = ON  → system permissive is healthy
        X17 = OFF → emergency stop or safety permissive has been lost

    Once X17 drops OFF, the PLC immediately:
        1. Latches the E-stop condition
        2. Stops the test
        3. Marks the system as internally stopped

    The latch is intentional. Even if the physical E-stop is released,
    the system remains stopped until the operator presses Reset.
*)

IF NOT X17 THEN
    SET(EStop_Latch);     (* Remember that an E-stop/permissive fault occurred *)
    RST(Run_Latch);       (* Immediately stop test execution *)
    SET(Internal_Stop);   (* Mark that the system was stopped by control logic *)
END_IF;

(*
    If the E-stop latch is active, force the machine to remain stopped.

    This protects against auto-restart after the physical permissive returns.
    The operator must intentionally reset the condition before restarting.
*)

IF EStop_Latch THEN
    RST(Run_Latch);
    SET(Internal_Stop);
END_IF;

(*
    Reset clears the E-stop latch only if the hardware permissive X17 is healthy.

    This prevents the operator from clearing the E-stop state while the actual
    safety/permissive input is still OFF.
*)

IF HMI_Reset AND X17 THEN
    RST(EStop_Latch);
END_IF;


(* ===================================================== *)
(* ---------------- TEST CHANGE ------------------------ *)
(* ===================================================== *)
(*
    Purpose:
    If the technician changes the selected test, the manager clears old state.

    This prevents "memory contamination" where internal bits from the previous
    test could affect the newly selected test.

    Example:
        If Test 1 left an internal actuator step bit ON,
        switching to Test 2 should not inherit that old state.

    This block resets:
        - run state
        - completion state
        - hold/failure states
        - cycle counters
        - failure counters
        - hold reason
        - known internal M bits used by individual tests
*)

IF (HMI_SelectedTest <> LastTest) THEN

    (* Stop the currently running test when the selected test changes *)
    RST(Run_Latch);

    (* Clear common cycle/completion state *)
    RST(Cycle_Complete);
    RST(Goal_Reached);
    RST(Internal_Stop);

    (* Clear next pause setting because it belongs to the previous test context *)
    NextPause_Setpoint := 0;

    (* Clear operator/admin/fault states *)
    RST(Hold_Latch);
    RST(Stopped_By_User);
    RST(Failure_State);
    RST(ConsecFailTrip);
    RST(HoldReason_Required);

    (* Reset production and failure counters for the newly selected test *)
    CycleCount := 0;
    FailtoRet := 0;
    ConsecutiveFailCnt := 0;
    HoldReason_Code := 0;

    (*
        Clear known internal memory bits used by test function blocks.

        These are explicit cleanup bits for legacy/test-specific logic.
        They prevent old sequence steps from carrying over into another test.
    *)
    RST(M20); RST(M21);
    RST(M22); RST(M23);
    RST(M32); RST(M33);

    RST(M6);  RST(M7);
    RST(M8);  RST(M9);
    RST(M10); RST(M16);

END_IF;

(*
    Store the current test selection so that the PLC can detect
    a selection change on the next scan.
*)

LastTest := HMI_SelectedTest;


(* ===================================================== *)
(* ---------------- RESET ------------------------------ *)
(* ===================================================== *)
(*
    Purpose:
    HMI_Reset returns the selected station/test to a clean starting state.

    Unlike changing tests, this reset does not change which test is selected.
    It clears the current test's progress, stops execution, and clears active
    hold/failure/completion states.

    This is the technician's normal way to recover from stopped/faulted states.
*)

IF HMI_Reset THEN

    (* Stop execution and clear common cycle/completion state *)
    RST(Run_Latch);
    RST(Cycle_Complete);
    RST(Goal_Reached);
    RST(Internal_Stop);

    (* Clear configured pause point *)
    NextPause_Setpoint := 0;

    (* Clear hold, stop, and failure-related states *)
    RST(Hold_Latch);
    RST(Stopped_By_User);
    RST(Failure_State);
    RST(ConsecFailTrip);
    RST(HoldReason_Required);

    (* Reset counters and hold reason *)
    CycleCount := 0;
    FailtoRet := 0;
    ConsecutiveFailCnt := 0;
    HoldReason_Code := 0;

END_IF;


(* ===================================================== *)
(* ---------------- HOLD / OFF-HOLD -------------------- *)
(* ===================================================== *)
(*
    Purpose:
    Hold allows a technician to intentionally pause/hold the station
    for a documented reason.

    This is different from:
        - Stop: technician stops the run
        - E-stop: safety/permissive fault
        - Failure: test/process fault
        - Complete: cycle target reached

    A hold should be traceable, especially for future Ignition/database logging.
*)

IF Hold_Rise THEN

    (*
        If the technician already selected a valid hold reason,
        immediately place the station on hold and stop the test.
    *)
    IF HoldReason_Valid THEN
        SET(Hold_Latch);
        RST(Run_Latch);

    (*
        If no valid hold reason was selected, do not place the station
        fully on hold yet. Instead, ask the HMI to prompt for a reason.
    *)
    ELSE
        SET(HoldReason_Required);
    END_IF;

END_IF;

(*
    If a hold was requested earlier but no reason was entered at that time,
    this block waits until the technician enters a valid reason.

    Once a valid reason appears, the station is placed on hold and
    the "reason required" prompt can be cleared.
*)

IF HoldReason_Required AND HoldReason_Valid THEN
    SET(Hold_Latch);
    RST(Run_Latch);
    RST(HoldReason_Required);
END_IF;

(*
    Off-hold releases the station from hold.

    It also clears the hold reason code so the next hold event requires
    a fresh reason instead of accidentally reusing the previous one.
*)

IF OffHold_Rise THEN

    RST(Hold_Latch);
    RST(HoldReason_Required);

    HoldReason_Code := 0;

END_IF;


(* ===================================================== *)
(* ---------------- START / STOP TOGGLE ---------------- *)
(* ===================================================== *)
(*
    Purpose:
    The HMI uses one combined Start/Stop command.

    On each rising edge:
        - If currently running → stop the test
        - If not running → attempt to start the test

    Starting is only allowed when all blocking conditions are clear.
*)

IF StartStop_Rise THEN

    (*
        If the test is currently running, the button acts as STOP.

        Stopped_By_User is set so the HMI can distinguish this from:
            - E-stop
            - failure
            - completion
            - hold
    *)
    IF Run_Latch THEN

        RST(Run_Latch);
        SET(Stopped_By_User);

    ELSE

        (*
            If the test is not currently running, the button acts as START.

            Start is allowed only if:
                - the test goal has not already been reached
                - no E-stop is latched
                - the hardware permissive X17 is healthy
                - station is not on hold
                - no active failure exists
                - the consecutive failure trip has not occurred
        *)
        IF (NOT Goal_Reached)
        AND (NOT EStop_Latch)
        AND X17
        AND (NOT Hold_Latch)
        AND (NOT Failure_State)
        AND (NOT ConsecFailTrip) THEN

            SET(Run_Latch);
            RST(Internal_Stop);
            RST(Stopped_By_User);

        END_IF;

    END_IF;

END_IF;


(* ===================================================== *)
(* ---------------- MANAGER EXECUTING ------------------ *)
(* ===================================================== *)
(*
    Purpose:
    Manager_Executing is the final "permission to run actual test logic."

    This is stronger than Run_Latch alone.

    Run_Latch means:
        "The operator requested running."

    Manager_Executing means:
        "The system is actually allowed to execute the selected test this scan."

    This prevents test function blocks from running during:
        - completed test
        - hold
        - failure
        - consecutive failure lockout
        - E-stop
        - lost hardware permissive
*)

Manager_Executing :=
Run_Latch
AND NOT Goal_Reached
AND NOT Hold_Latch
AND NOT Failure_State
AND NOT ConsecFailTrip
AND X17
AND NOT EStop_Latch;


(* ===================================================== *)
(* ---------------- EXECUTION -------------------------- *)
(* ===================================================== *)
(*
    Purpose:
    Clear the pass/fail pulse interface before calling the selected test.

    These pulse bits represent the result of one completed cycle.
    They must not remain ON from a previous scan or previous test.
*)

CyclePass_Pulse := FALSE;
CycleFail_Pulse := FALSE;

(*
    Only execute test logic if the manager says it is safe and allowed.

    Exactly one selected test function block is called based on HMI_SelectedTest.
    Each test function block may output:
        - CyclePass_Pulse
        - CycleFail_Pulse

    The manager copies those pulses into common global pulse labels so that
    downstream failure tracking can work the same way for every test.
*)

IF Manager_Executing THEN

    CASE HMI_SelectedTest OF

        1:
            (* Trim cycling test *)
            TrimInst();
            CyclePass_Pulse := TrimInst.CyclePass_Pulse;
            CycleFail_Pulse := TrimInst.CycleFail_Pulse;

        2:
            (*
                Exit test.

                HMI_UseCloser allows the technician/HMI to select whether
                closer-related behavior should be included in this exit test.
            *)
            ExitInst(
                UseCloser := HMI_UseCloser
            );
            CyclePass_Pulse := ExitInst.CyclePass_Pulse;
            CycleFail_Pulse := ExitInst.CycleFail_Pulse;

        4:
            (* QEL cycling test *)
            QELInst();
            CyclePass_Pulse := QELInst.CyclePass_Pulse;
            CycleFail_Pulse := QELInst.CycleFail_Pulse;

        5:
            (* Mag lock cycling test *)
            MagLockInst();
            CyclePass_Pulse := MagLockInst.CyclePass_Pulse;
            CycleFail_Pulse := MagLockInst.CycleFail_Pulse;

        6:
            (* Chexit cycling test *)
            ChexitInst();
            CyclePass_Pulse := ChexitInst.CyclePass_Pulse;
            CycleFail_Pulse := ChexitInst.CycleFail_Pulse;

    END_CASE;

END_IF;


(* ===================================================== *)
(* ---------------- FAILURE TRACKING ------------------- *)
(* ===================================================== *)
(*
    Purpose:
    Track both total failures and consecutive failures.

    Total failures:
        FailtoRet

    Consecutive failures:
        ConsecutiveFailCnt

    A pass resets the consecutive failure counter.
    A failure increments both total failures and consecutive failures.

    This allows the system to detect not just "how many failures happened,"
    but whether failures are happening repeatedly in a row.
*)

IF CyclePass_Pulse AND NOT CycleFail_Pulse THEN

    (* A successful cycle breaks the failure streak *)
    ConsecutiveFailCnt := 0;

END_IF;

IF CycleFail_Pulse AND NOT CyclePass_Pulse THEN

    (* A failed cycle increments both total failures and current failure streak *)
    FailtoRet := FailtoRet + 1;
    ConsecutiveFailCnt := ConsecutiveFailCnt + 1;

END_IF;

(*
    A cycle cannot logically be both pass and fail in the same scan.

    If both pulses are ON at the same time, the selected test FB has produced
    contradictory information. Treat this as a logic fault and stop trusting
    the running condition.
*)

IF CyclePass_Pulse AND CycleFail_Pulse THEN

    SET(Failure_State);

END_IF;

(*
    If five failures occur consecutively, trip the failure lockout.

    This stops the test and requires operator/technician intervention.
*)

IF ConsecutiveFailCnt >= 5 THEN

    SET(ConsecFailTrip);
    SET(Failure_State);
    SET(Internal_Stop);
    RST(Run_Latch);

END_IF;


(* ===================================================== *)
(* ---------------- CPM / COMPLETION / NEXT PAUSE ------- *)
(* ===================================================== *)
(*
    Purpose:
    Run utility calculations after test execution.

    These utility function blocks do not perform the physical test sequence.
    They calculate status information used by the HMI and future database logging.
*)

(*
    CPMInst calculates cycles per minute.

    It uses:
        - current cycle count
        - whether the manager is actually executing

    If the manager is not executing, CPM calculation should not treat idle time
    as normal test time unless CPMInst is intentionally designed to do so.
*)

CPMInst(
    CycleCount_In := CycleCount,
    Enable := Manager_Executing
);

(*
    COMPInst estimates the projected completion date/time.

    It uses:
        - current cycle count
        - target cycle count
        - calculated CPM

    Important:
        This is an estimate, not a guaranteed completion time.
        If CPM changes, the displayed completion date/time may move.
*)

COMPInst(
    CycleCount_In  := CycleCount,
    CycleTarget_In := CycleTarget,
    CPM_In         := CPMInst.CPM
);

(*
    NextPauseInst checks whether the station has reached the next configured
    pause point based on cycle count.

    Example:
        If NextPause_Setpoint = 1000,
        the manager can stop the test after the cycle that reaches 1000.
*)

NextPauseInst(
    CycleCount_In := CycleCount,
    Setpoint      := NextPause_Setpoint,
    Reset         := HMI_Reset
);

(*
    Copy utility outputs to HMI/global labels.

    These are the values the HMI displays and Ignition may eventually read.
*)

HMI_CPM := CPMInst.CPM;

HMI_Comp_Year  := COMPInst.Comp_Year;
HMI_Comp_Month := COMPInst.Comp_Month;
HMI_Comp_Day   := COMPInst.Comp_Day;
HMI_Comp_Hour  := COMPInst.Comp_Hour;
HMI_Comp_Min   := COMPInst.Comp_Min;

NextPause_Active := NextPauseInst.Trigger;

(*
    If the Next Pause point is reached at the same time a cycle completes,
    stop the test and mark it as a user-style stop.

    This stops between completed cycles instead of stopping mid-cycle.
*)

IF NextPauseInst.Trigger AND Cycle_Complete THEN
    RST(Run_Latch);
    SET(Stopped_By_User);
END_IF;


(* ===================================================== *)
(* ---------------- COMPLETION ------------------------- *)
(* ===================================================== *)
(*
    Purpose:
    If the target cycle count has been reached, stop the test and mark
    the system internally stopped.

    This is a normal completion stop, not a user stop.
*)

IF Goal_Reached THEN
    RST(Run_Latch);
    RST(Stopped_By_User);
    SET(Internal_Stop);
END_IF;


(* ===================================================== *)
(* ---------------- DERIVED STATES --------------------- *)
(* ===================================================== *)
(*
    Purpose:
    Convert the internal manager bits into clean HMI states.

    These states are easier for technicians, HMI screens, and Ignition tags
    to understand than raw internal PLC latches.

    Important:
    These are derived from the logic above. They should not be driven directly
    by the HMI.
*)

(*
    Running_State means the system is truly executing a test this scan,
    not merely that the operator pressed Start.
*)

Running_State :=
Manager_Executing;

(*
    Held_State means the station is intentionally on hold.
*)

Held_State :=
Hold_Latch;

(*
    Paused_State means the operator stopped the test with the Start/Stop
    button and the station is otherwise healthy.

    This excludes:
        - completed test
        - hold
        - failure
        - consecutive failure lockout
        - E-stop
        - lost hardware permissive
*)

Paused_State :=
Stopped_By_User
AND NOT Run_Latch
AND NOT Goal_Reached
AND NOT Hold_Latch
AND NOT Failure_State
AND NOT ConsecFailTrip
AND X17
AND NOT EStop_Latch;

(*
    Idle_State means the station is not running, not held, not paused,
    not failed, not complete, and not in E-stop.

    This is the clean waiting state.
*)

Idle_State :=
NOT Run_Latch
AND NOT Stopped_By_User
AND NOT Goal_Reached
AND NOT Hold_Latch
AND NOT Paused_State
AND NOT Failure_State
AND NOT EStop_Latch;

(*
    Test_Complete drives the HMI complete lamp.
*)

Test_Complete := Goal_Reached;

(*
    HMI_EStop drives the HMI e-stop indication.

    It uses the latched E-stop state, not only raw X17,
    so the HMI continues to show the event until reset.
*)

HMI_EStop := EStop_Latch;
