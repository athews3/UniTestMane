(* ===================================================== *)
(* ---------------- TEST CHANGE ------------------------- *)
(* ===================================================== *)
IF (HMI_SelectedTest <> LastTest) THEN

    (* --- stop execution --- *)
    RST(Run_Latch);

    (* --- reset core state --- *)
    RST(Cycle_Complete);
    RST(Goal_Reached);
    RST(Internal_Stop);
	RST(NextPause_Setpoint);
    (* --- reset pause --- *)
    RST(Paused_Latch);

    (* --- reset counters --- *)
    CycleCount := 0;
    FailtoRet := 0;

    (* -------- MEMORY CONTAMINATION CONTROL -------- *)
    RST(M20); RST(M21);
    RST(M22); RST(M23);
    RST(M32); RST(M33);

    RST(M6);  RST(M7);
    RST(M8);  RST(M9);
    RST(M10); RST(M16);

END_IF;

LastTest := HMI_SelectedTest;


(* ===================================================== *)
(* ---------------- START (EDGE) ------------------------ *)
(* ===================================================== *)
IF (HMI_Start AND NOT PrevStart AND NOT Paused_Latch AND NOT Goal_Reached AND X17 AND NOT EStop_Latch) THEN
    SET(Run_Latch);
    RST(Internal_Stop);
END_IF;

PrevStart := HMI_Start;


(* ===================================================== *)
(* ---------------- STOP ------------------------------- *)
(* ===================================================== *)
IF HMI_Stop THEN
    RST(Run_Latch);
    RST(Paused_Latch);
    SET(Internal_Stop);
END_IF;

(* ===================================================== *)
(* ---------------- RESET ------------------------------ *)
(* ===================================================== *)
IF HMI_Reset THEN
    RST(Run_Latch);
    RST(Cycle_Complete);
    RST(Goal_Reached);
    RST(Paused_Latch);
    RST(Internal_Stop);
	RST(NextPause_Setpoint);
    CycleCount := 0;
    FailtoRet := 0;
END_IF;


(* ===================================================== *)
(* ---------------- PAUSE / RESUME --------------------- *)
(* ===================================================== *)

(*Keep pause/resume latch actuated when button is released*)

Pause_Rise := (HMI_Pause AND NOT PrevPause);
Resume_Rise := (HMI_Resume AND NOT PrevResume);

PrevPause := HMI_Pause;
PrevResume := HMI_Resume;

IF (Pause_Rise AND NOT Paused_Latch) THEN
    SET(Paused_Latch);
    RST(Run_Latch);
END_IF;

IF (Resume_Rise AND Paused_Latch AND NOT Goal_Reached AND X17 AND NOT EStop_Latch) THEN
    RST(Paused_Latch);
    SET(Run_Latch);
END_IF;


(* ===================================================== *)
(* ---------------- INTERLOCK -------------------------- *)
(* ===================================================== *)
(*Trip*)
IF NOT X17 THEN
    RST(Run_Latch);
    SET(Internal_Stop);
END_IF;

(*Enforce stop*)

IF EStop_Latch THEN
	RST(Run_Latch);
	SET(Internal_Stop);
	
(*HMI Operator reset*)
IF HMI_Reset AND X17 THEN
	RST(EStop_Latch);
END_IF;

(* ===================================================== *)
(* ---------------- RESERVATION ------------------------- *)
(* ===================================================== *)

(* --- EDGE DETECTION --- *)
Reserve_Rise   := (Reserve_CMD AND NOT PrevReserve);
ClearRes_Rise  := (ClearReserve_CMD AND NOT PrevClearRes);

PrevReserve  := Reserve_CMD;
PrevClearRes := ClearReserve_CMD;


(* --- TIMER (state observer only) --- *)
ReserveTimer(
IN := Reserve_Rise,
PT := T#5H
);


(* --- SET (ownership acquisition) --- *)
IF (Reserve_Rise AND NOT Reserved_Latch) THEN
    SET(Reserved_Latch);
    TechInit := HMI_Init;
END_IF;


(* --- CLEAR (explicit user override - highest priority) --- *)
IF ClearRes_Rise THEN
    RST(Reserved_Latch);
    TechInit := 0;
END_IF;


(* --- TIME EXPIRATION (automatic revoke) --- *)
IF (Reserved_Latch AND ReserveTimer.Q) THEN
    RST(Reserved_Latch);
    TechInit := 0;
END_IF;

(* ===================================================== *)
(* ---------------- EXECUTION -------------------------- *)
(* ===================================================== *)
IF (Run_Latch AND NOT Goal_Reached AND NOT Paused_Latch) THEN

    CASE HMI_SelectedTest OF
        1: TrimInst();
        2: ExitAirInst();
        3: ExitCloserInst();
        4: QELInst();
        5: MagLockInst();
        6: ChexitInst();
    END_CASE;

END_IF;

(*Cycles per minute*)
CPMInst(
CycleCount_IN := CycleCount,
Enable := Running_State);

(*COMPLETION*)
COMPInst(
CycleCount_In := CycleCount,
CycleTarget_In := CycleTarget,
CPM := CPMInst.CPM

);
(*Next Pause*)
NextPauseInst(
CycleCount_In := CycleCount,
Setpoint := NextPause_Setpoint,
Reset := HMI_Reset);

(*Connect to HMI and establish completion date/time*)
HMI_CPM := CPMInst.CPM;

HMI_Comp_Year := COMPInst.Comp_Year;
HMI_Comp_Month := COMPInst.Comp_Month;
HMI_Comp_Day := COMPInst.Comp_Day;
HMI_Comp_Hour := COMPInst.Comp_Hour;
HMI_Comp_Min := COMPInst.Comp_Min;

NextPause_Active := NextPAUSEInst.Trigger;

(* ===================================================== *)
(* ---------------- COMPLETION ------------------------- *)
(* ===================================================== *)
IF Goal_Reached THEN
    RST(Run_Latch);
    SET(Internal_Stop);
END_IF;


(* ===================================================== *)
(* ---------------- DERIVED STATES --------------------- *)
(* ===================================================== *)

Running_State := Run_Latch AND NOT Goal_Reached;
Paused_State  := Paused_Latch;
Reserved_State := Reserved_Latch;

Idle_State :=
(NOT Run_Latch)
AND (NOT Paused_Latch)
AND (NOT Goal_Reached)
AND (Internal_Stop OR NOT X17);

HMI_Estop := EStop_Latch;

(* ===================================================== *)
(* ---------------- HMI LAMPS -------------------------- *)
(* ===================================================== *)

Test_Running  := Running_State;
Test_Complete := Goal_Reached;
Test_Ready    := Idle_State;
