(* ---------------- TEST CHANGE ---------------- *)
IF (D100 <> LastTest) THEN
    RST(M4);
    RST(M1);
    RST(M2);
    RST(M205);
    D200 := 0;
END_IF;

LastTest := D100;


(* ---------------- START ---------------- *)
IF (M200 AND NOT PrevStart AND NOT M210 AND NOT M2 AND X17) THEN
    SET(M4);
    RST(M205);
END_IF;
PrevStart := M200;


(* ---------------- STOP ---------------- *)
IF M201 THEN
    RST(M4);
    RST(M210);
    SET(M205);
END_IF;


(* ---------------- RESET ---------------- *)
IF M202 THEN
    RST(M4);
    RST(M1);
    RST(M2);
    RST(M210);
    RST(M205);
    D200 := 0;
END_IF;


(* ---------------- PAUSE ---------------- *)
IF (M210_CMD AND NOT M210) THEN
    SET(M210);
    RST(M4);
END_IF;

IF (M211_CMD AND M210 AND NOT M2 AND X17) THEN
    RST(M210);
    SET(M4);
END_IF;


(* ---------------- INTERLOCK ---------------- *)
IF NOT X17 THEN
    RST(M4);
    SET(M205);
END_IF;


(* ---------------- RESERVATION ---------------- *)
IF (M220_CMD AND NOT M220) THEN
    D300 := D110;                      (* initials *)
    D301 := SD520 + 3600;
    SET(M220);
END_IF;

IF M221_CMD THEN
    RST(M220);
    D300 := 0;
END_IF;

IF (M220 AND SD520 > D301) THEN
    RST(M220);
    D300 := 0;
END_IF;


(* ---------------- EXECUTION ---------------- *)
IF (M4 AND NOT M2 AND NOT M210) THEN

    CASE D100 OF
        1: TrimInst();
        2: ExitAirInst();
        3: ExitCloserInst();
        4: QELInst();
        5: MaglockInst();
        6: ChexitInst();
    END_CASE;

END_IF;


(* ---------------- COMPLETION ---------------- *)
IF M2 THEN
    RST(M4);
    SET(M205);
END_IF;


(* ---------------- DERIVED STATES ---------------- *)
M300 := M4 AND NOT M2;
M301 := M210;
M303 := M220;

M302 :=
    (NOT M4)
    AND (NOT M210)
    AND (NOT M2)
    AND (M205 OR NOT X17);
