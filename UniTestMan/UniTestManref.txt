(* ===================================================== *)
(* ---------------- TEST CHANGE ------------------------- *)
(* ===================================================== *)
IF (HMI_SelectedTest <> LastTest) THEN

    (* --- stop execution --- *)
    RST(Run_Latch);

    (* --- reset core state --- *)
    RST(Cycle_Complete);
    RST(Goal_Reached);
    RST(Internal_Stop);

    (* --- reset pause --- *)
    RST(Paused_Latch);

    (* --- reset counters --- *)
    CycleCount := 0;
    FailtoRet := 0;

    (* -------- MEMORY CONTAMINATION CONTROL -------- *)
    RST(M20); RST(M21);
    RST(M22); RST(M23);
    RST(M32); RST(M33);

    RST(M6);  RST(M7);
    RST(M8);  RST(M9);
    RST(M10); RST(M16);

END_IF;

LastTest := HMI_SelectedTest;


(* ===================================================== *)
(* ---------------- START (EDGE) ------------------------ *)
(* ===================================================== *)
IF (HMI_Start AND NOT PrevStart AND NOT Paused_Latch AND NOT Goal_Reached AND X17) THEN
    SET(Run_Latch);
    RST(Internal_Stop);
END_IF;

PrevStart := HMI_Start;


(* ===================================================== *)
(* ---------------- STOP ------------------------------- *)
(* ===================================================== *)
IF HMI_Stop THEN
    RST(Run_Latch);
    RST(Paused_Latch);
    SET(Internal_Stop);
END_IF;


(* ===================================================== *)
(* ---------------- RESET ------------------------------ *)
(* ===================================================== *)
IF HMI_Reset THEN
    RST(Run_Latch);
    RST(Cycle_Complete);
    RST(Goal_Reached);
    RST(Paused_Latch);
    RST(Internal_Stop);

    CycleCount := 0;
    FailtoRet := 0;
END_IF;


(* ===================================================== *)
(* ---------------- PAUSE / RESUME --------------------- *)
(* ===================================================== *)
IF (Pause_CMD AND NOT Paused_Latch) THEN
    SET(Paused_Latch);
    RST(Run_Latch);
END_IF;

IF (Resume_CMD AND Paused_Latch AND NOT Goal_Reached AND X17) THEN
    RST(Paused_Latch);
    SET(Run_Latch);
END_IF;


(* ===================================================== *)
(* ---------------- INTERLOCK -------------------------- *)
(* ===================================================== *)
IF NOT X17 THEN
    RST(Run_Latch);
    SET(Internal_Stop);
END_IF;


(* ===================================================== *)
(* ---------------- RESERVATION ------------------------ *)
(* ===================================================== *)
IF (Reserve_CMD AND NOT Reserved_Latch) THEN
    TechnicianInitials := HMI_Initials;   (* labeled, not raw D *)
    ReservationExpiry  := SD520 + 3600;
    SET(Reserved_Latch);
END_IF;

IF ClearReserve_CMD THEN
    RST(Reserved_Latch);
    TechnicianInitials := 0;
END_IF;

IF (Reserved_Latch AND SD520 > ReservationExpiry) THEN
    RST(Reserved_Latch);
    TechnicianInitials := 0;
END_IF;


(* ===================================================== *)
(* ---------------- EXECUTION -------------------------- *)
(* ===================================================== *)
IF (Run_Latch AND NOT Goal_Reached AND NOT Paused_Latch) THEN

    CASE HMI_SelectedTest OF
        1: TrimInst();
        2: ExitAirInst();
        3: ExitCloserInst();
        4: QELInst();
        5: MagLockInst();
        6: ChexitInst();
    END_CASE;

END_IF;


(* ===================================================== *)
(* ---------------- COMPLETION ------------------------- *)
(* ===================================================== *)
IF Goal_Reached THEN
    RST(Run_Latch);
    SET(Internal_Stop);
END_IF;


(* ===================================================== *)
(* ---------------- DERIVED STATES --------------------- *)
(* ===================================================== *)

Running_State := Run_Latch AND NOT Goal_Reached;
Paused_State  := Paused_Latch;
Reserved_State := Reserved_Latch;

Idle_State :=
    (NOT Run_Latch)
    AND (NOT Paused_Latch)
    AND (NOT Goal_Reached)
    AND (Internal_Stop OR NOT X17);


(* ===================================================== *)
(* ---------------- HMI LAMPS -------------------------- *)
(* ===================================================== *)

Test_Running  := Running_State;
Test_Complete := Goal_Reached;
Test_Ready    := Idle_State;
