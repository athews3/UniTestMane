(* ===================================================== *)
(* ---------------- EDGE DETECTION --------------------- *)
(* ===================================================== *)

StartStop_Rise := HMI_StartStop AND NOT PrevStartStop;
Hold_Rise      := HMI_Hold      AND NOT PrevHold;
OffHold_Rise   := HMI_OffHold   AND NOT PrevOffHold;

PrevStartStop := HMI_StartStop;
PrevHold      := HMI_Hold;
PrevOffHold   := HMI_OffHold;


(* ===================================================== *)
(* ---------------- HOLD REASON VALIDATION ------------- *)
(* ===================================================== *)

HoldReason_Valid := (HoldReason_Code > 0);


(* ===================================================== *)
(* ---------------- SAFETY / INTERLOCK ----------------- *)
(* ===================================================== *)

IF NOT X17 THEN
    RST(Run_Latch);
    SET(Internal_Stop);
END_IF;

IF EStop_Latch THEN
    RST(Run_Latch);
    SET(Internal_Stop);
END_IF;

IF HMI_Reset AND X17 THEN
    RST(EStop_Latch);
END_IF;


(* ===================================================== *)
(* ---------------- TEST CHANGE ------------------------ *)
(* ===================================================== *)

IF (HMI_SelectedTest <> LastTest) THEN

    RST(Run_Latch);

    RST(Cycle_Complete);
    RST(Goal_Reached);
    RST(Internal_Stop);
    RST(NextPause_Setpoint);

    RST(Hold_Latch);
    RST(Stopped_By_User);
    RST(Failure_State);
    RST(ConsecFailTrip);
    RST(HoldReason_Required);

    CycleCount := 0;
    FailtoRet := 0;
    ConsecutiveFailCnt := 0;
    HoldReason_Code := 0;

    RST(M20); RST(M21);
    RST(M22); RST(M23);
    RST(M32); RST(M33);

    RST(M6);  RST(M7);
    RST(M8);  RST(M9);
    RST(M10); RST(M16);

END_IF;

LastTest := HMI_SelectedTest;


(* ===================================================== *)
(* ---------------- RESET ------------------------------ *)
(* ===================================================== *)

IF HMI_Reset THEN

    RST(Run_Latch);
    RST(Cycle_Complete);
    RST(Goal_Reached);
    RST(Internal_Stop);
    RST(NextPause_Setpoint);

    RST(Hold_Latch);
    RST(Stopped_By_User);
    RST(Failure_State);
    RST(ConsecFailTrip);
    RST(HoldReason_Required);

    CycleCount := 0;
    FailtoRet := 0;
    ConsecutiveFailCnt := 0;
    HoldReason_Code := 0;

END_IF;


(* ===================================================== *)
(* ---------------- HOLD / OFF-HOLD -------------------- *)
(* ===================================================== *)

IF Hold_Rise THEN

    IF HoldReason_Valid THEN
        SET(Hold_Latch);
        RST(Run_Latch);
    ELSE
        SET(HoldReason_Required);
    END_IF;

END_IF;

IF HoldReason_Required AND HoldReason_Valid THEN
    SET(Hold_Latch);
    RST(Run_Latch);
    RST(HoldReason_Required);
END_IF;

IF OffHold_Rise THEN

    RST(Hold_Latch);
    RST(HoldReason_Required);

    HoldReason_Code := 0;

END_IF;


(* ===================================================== *)
(* ---------------- START / STOP TOGGLE ---------------- *)
(* ===================================================== *)

IF StartStop_Rise THEN

    IF Run_Latch THEN

        RST(Run_Latch);
        SET(Stopped_By_User);

    ELSE

        IF (NOT Goal_Reached)
        AND (NOT EStop_Latch)
        AND X17
        AND (NOT Hold_Latch)
        AND (NOT Failure_State)
        AND (NOT ConsecFailTrip) THEN

            SET(Run_Latch);
            RST(Internal_Stop);
            RST(Stopped_By_User);

        END_IF;

    END_IF;

END_IF;


(* ===================================================== *)
(* ---------------- MANAGER EXECUTING ------------------ *)
(* ===================================================== *)

Manager_Executing :=
Run_Latch
AND NOT Goal_Reached
AND NOT Hold_Latch
AND NOT Failure_State
AND NOT ConsecFailTrip
AND X17
AND NOT EStop_Latch;


(* ===================================================== *)
(* ---------------- EXECUTION -------------------------- *)
(* ===================================================== *)

IF Manager_Executing THEN

    CASE HMI_SelectedTest OF

        1:
            TrimInst();

        2:
            ExitInst(
                UseCloser := HMI_UseCloser
            );

        3:
            ExitCloserInst();
            (* temporary legacy mapping *)

        4:
            QELInst();

        5:
            MagLockInst();

        6:
            ChexitInst();

    END_CASE;

END_IF;


(* ===================================================== *)
(* ---------------- FAILURE TRACKING ------------------- *)
(* ===================================================== *)

IF CycleFail_Pulse THEN

    FailtoRet := FailtoRet + 1;
    ConsecutiveFailCnt := ConsecutiveFailCnt + 1;

END_IF;

IF CyclePass_Pulse THEN

    ConsecutiveFailCnt := 0;

END_IF;

IF ConsecutiveFailCnt >= 5 THEN

    SET(ConsecFailTrip);
    SET(Failure_State);
    SET(Internal_Stop);
    RST(Run_Latch);

END_IF;


(* ===================================================== *)
(* ---------------- CPM / COMPLETION / NEXT PAUSE ------- *)
(* ===================================================== *)

CPMInst(
    CycleCount_IN := CycleCount,
    Enable := Manager_Executing
);

COMPInst(
    CycleCount_In  := CycleCount,
    CycleTarget_In := CycleTarget,
    CPM            := CPMInst.CPM
);

NextPauseInst(
    CycleCount_In := CycleCount,
    Setpoint      := NextPause_Setpoint,
    Reset         := HMI_Reset
);

HMI_CPM := CPMInst.CPM;

HMI_Comp_Year  := COMPInst.Comp_Year;
HMI_Comp_Month := COMPInst.Comp_Month;
HMI_Comp_Day   := COMPInst.Comp_Day;
HMI_Comp_Hour  := COMPInst.Comp_Hour;
HMI_Comp_Min   := COMPInst.Comp_Min;

NextPause_Active := NextPauseInst.Trigger;

IF NextPauseInst.Trigger AND Cycle_Complete THEN
    RST(Run_Latch);
    SET(Stopped_By_User);
END_IF;


(* ===================================================== *)
(* ---------------- COMPLETION ------------------------- *)
(* ===================================================== *)

IF Goal_Reached THEN
    RST(Run_Latch);
    RST(Stopped_By_User);
    SET(Internal_Stop);
END_IF;


(* ===================================================== *)
(* ---------------- DERIVED STATES --------------------- *)
(* ===================================================== *)

Running_State :=
Manager_Executing;

Complete_State :=
Goal_Reached;

Held_State :=
Hold_Latch;

Paused_State :=
Stopped_By_User
AND NOT Run_Latch
AND NOT Goal_Reached
AND NOT Hold_Latch
AND NOT Failure_State
AND NOT ConsecFailTrip
AND X17
AND NOT EStop_Latch;

Failure_State :=
ConsecFailTrip
OR Internal_Stop;

Idle_State :=
NOT Run_Latch
AND NOT Stopped_By_User
AND NOT Goal_Reached
AND NOT Hold_Latch
AND NOT Paused_State;

Reserved_State := Held_State;

HMI_Estop := EStop_Latch;


(* ===================================================== *)
(* ---------------- HMI LAMPS -------------------------- *)
(* ===================================================== *)

Test_Running  := Running_State;
Test_Complete := Complete_State;
Test_Ready    := Idle_State;
Test_Paused   := Paused_State;
Test_Held     := Held_State;
Test_Reserved := Held_State;
Test_Faulted  := Failure_State;
