(* ======================================= *)
(* Universal Test Manager *)
(* ======================================= *)

(* ---------- One-time initialization ---------- *)
IF ( D101 = -1 ) THEN
    D101 := D100;
END_IF;


(* ---------- Reset when test selection changes ---------- *)
IF ( D100 <> D101 ) THEN

    (* Reset common test state *)
    RST(M1);        (* Main Break *)
    RST(M2);        (* Cycle Goal *)
    RST(M101);      (* Output latch *)
    RST(M102);      (* Output latch *)
    RST(M4);        (* Start/Enable *)

    (* Reset timers *)
    T0 := FALSE;
    T1 := FALSE;
    T2 := FALSE;

    (* Reset counters *)
    D200 := 0;

END_IF;

D101 := D100;   (* Remember last selected test *)


(* ---------- Default outputs OFF (safe) ---------- *)
Y0 := FALSE;
Y1 := FALSE;
Y2 := FALSE;


(* ================= CASE DISPATCH ================= *)
CASE D100 OF

    0:  (* ---------------- IDLE ---------------- *)
	(* Outputs remain off *)
	;

    1:  (* ----------- TRIM CYCLE --------------- *)

	IF ( (X0 OR T1) AND X17 AND (NOT M1) AND (NOT M2) AND M4 ) THEN
		T1 := TRUE;
        ELSE
		T1 := FALSE;
	END_IF;

	IF ( T1 AND X17 AND (NOT M1) ) THEN
		Y0 := TRUE;
		T2 := TRUE;
        ELSE
		T2 := FALSE;
	END_IF;
	
	IF ( T2 AND X17 AND (NOT X0) ) THEN
		SET(M1);
		D200 := D200 + 1;
	END_IF;

	IF ( D200 > D202 ) THEN
		SET(M2);
	END_IF;

    2:  (* ----- EXIT CYCLE – AIR CYLINDER ------- *)

	IF ( X0 AND (NOT X1) AND X17 AND (NOT M1) AND (NOT M2) AND M4 ) THEN
		T0 := TRUE;
        ELSE
		T0 := FALSE;
	END_IF;
	
	IF ( T0 AND X17 AND (NOT M102) ) THEN
		SET(M101);
	END_IF;

	IF ( M102 OR (NOT X17) ) THEN
		RST(M101);
	END_IF;

	T1 := M101;
	
	IF ( T1 AND X1 AND X17 AND (NOT M1) ) THEN
		SET(M102);
	END_IF;

	IF ( M1 OR (NOT X17) ) THEN
		RST(M102);
	END_IF;

	T2 := M102;
	

	Y1 := M101;
	Y2 := M102;

	IF ( T2 AND X17 ) THEN
		SET(M1);
		D200 := D200 + 1;
	END_IF;

	IF ( D200 > D202 ) THEN
		SET(M2);
	END_IF;

    3:  (* ---- EXIT CYCLE – CLOSER --------------- *)

	IF ( X0 AND (NOT X1) AND X17 AND (NOT M1) AND (NOT M2) AND M4 ) THEN
		T0 := TRUE;
        ELSE
		T0 := FALSE;
	END_IF;
	
	IF ( T0 AND X17 AND (NOT M102) ) THEN
		SET(M101);
	END_IF;

	IF ( M102 OR (NOT X17) ) THEN
		RST(M101);
	END_IF;

	T1 := M101;
	

	IF ( T1 AND X1 AND X17 AND (NOT M1) ) THEN
		SET(M102);
	END_IF;

	IF ( M1 OR (NOT X17) ) THEN
		RST(M102);
	END_IF;

	T2 := M102;
	
	Y1 := M101;
	Y2 := M102;

	IF ( T2 AND X17 ) THEN
		SET(M1);
		D200 := D200 + 1;
	END_IF;

	IF ( D200 > D202 ) THEN
		SET(M2);
	END_IF;

    ELSE
	(* Undefined test ID – fail safe *)
	;

END_CASE;


(* ---------- Heartbeat ---------- *)
IF ( (NOT X17) AND FX5CPU.stSM.b1sClock ) THEN
    SET(M3);
	ELSE
    RST(M3);
END_IF;


(* ---------- E‑Stop handling ---------- *)
IF ( NOT X17 ) THEN
    RST(M4);
END_IF;
