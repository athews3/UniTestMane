(* Pre-open delay with self-hold *)
tPreDelay(IN := ((tPreDelay.Q OR X0) AND NOT X1 AND X17 AND NOT M1 AND NOT M2 AND M4), PT := T#1.5s);


(* OPEN -ends when tCloseHold duration ends *)
IF (tPreDelay.Q AND X17 AND NOT mClose AND NOT X1 AND M4) THEN
    mOpen := TRUE;
ELSE
	mOpen := FALSE;
END_IF;

tCloseHold(IN := X1, PT :=T#750ms);

(* CLOSE (corrected: no coexistence & tied to tCloseHold *)
IF (tCloseHold.Q AND X17 AND M4) THEN
    mClose := TRUE;
ELSE
	mClose := FALSE;
END_IF;

(* Outputs *)
Y1 := mOpen;
Y2 := mClose;


(* Cycle complete  *)
IF (mClose AND NOT prevClose AND X17 AND M4) THEN
    SET(M1);
    CycleCount := CycleCount + 1;
END_IF;

PrevClose := mClose;

(* Goal reached *)
IF (CycleCount >= CycleTarget) AND (CycleTarget > 0)THEN
    SET(M2);
END_IF;
