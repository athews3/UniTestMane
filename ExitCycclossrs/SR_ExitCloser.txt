(* ---------------- Pre-Open Delay ---------------- *)
tPreDelay(
    IN := (X0 AND NOT X1 AND X17 AND NOT M1 AND NOT M2 AND M4),
    PT := T#1s
);


(* ---------------- Pushbar OPEN ---------------- *)
tOpenHold(
    IN := ((tPreDelay.Q AND X17 AND NOT M102 AND M4) OR M101),
    PT := T#400ms
);

IF (tOpenHold.Q AND M4) THEN
    M101 := TRUE;
ELSE
    IF (NOT tPreDelay.Q OR NOT M4 OR M102) THEN
        M101 := FALSE;
    END_IF;
END_IF;


(* ---------------- CLOSE (Closer Action) ---------------- *)
tCloseHold(
    IN := ((M101 AND X1 AND X17 AND NOT M1 AND M4) OR M102),
    PT := T#2.3s
);

IF (tCloseHold.Q AND M4) THEN
    M102 := TRUE;
ELSE
    IF (NOT M101 OR NOT M4) THEN
        M102 := FALSE;
    END_IF;
END_IF;


(* ---------------- Outputs ---------------- *)
Y1 := M101;
Y2 := M102;


(* ---------------- Cycle Complete ---------------- *)
IF (tCloseHold.Q AND X17 AND M4) THEN
    SET(M1);
    CycleCount := CycleCount + 1;
END_IF;


(* ---------------- Goal Reached ---------------- *)
IF (CycleCount >= CycleTarget) AND (CycleTarget > 0) THEN
    SET(M2);
END_IF;
